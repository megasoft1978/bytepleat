cmake_minimum_required(VERSION 3.24)
project(BytepleatExplorerCommands LANGUAGES CXX)

if(NOT WIN32)
  message(FATAL_ERROR "The Explorer command DLL can only be built on Windows.")
endif()

add_library(BytepleatExplorerCommands SHARED src/ExplorerCommands.cpp src/BytepleatExplorerCommands.def)
target_compile_features(BytepleatExplorerCommands PRIVATE cxx_std_17)
target_compile_definitions(BytepleatExplorerCommands PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(BytepleatExplorerCommands PRIVATE "$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
target_link_libraries(BytepleatExplorerCommands PRIVATE ole32 shell32 shlwapi uuid)
set_target_properties(BytepleatExplorerCommands PROPERTIES OUTPUT_NAME "BytepleatExplorerCommands")

include(CTest)
if(BUILD_TESTING)
  add_executable(BytepleatExplorerCommandsTests tests/ExplorerCommandsTests.cpp)
  target_compile_features(BytepleatExplorerCommandsTests PRIVATE cxx_std_17)
  target_compile_definitions(BytepleatExplorerCommandsTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(BytepleatExplorerCommandsTests PRIVATE "$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
  target_link_libraries(BytepleatExplorerCommandsTests PRIVATE shell32 ole32 uuid)
  add_dependencies(BytepleatExplorerCommandsTests BytepleatExplorerCommands)
  # Invoked only after MSIX installation by Invoke-PackagedSmoke.ps1, not CTest.
  if(CMAKE_GENERATOR_PLATFORM MATCHES "^[Aa][Rr][Mm]64$")
    add_executable(BytepleatPackagedComSmoke tests/PackagedComSmoke.cpp)
    target_compile_features(BytepleatPackagedComSmoke PRIVATE cxx_std_17)
    target_compile_definitions(BytepleatPackagedComSmoke PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(BytepleatPackagedComSmoke PRIVATE "$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
    target_link_libraries(BytepleatPackagedComSmoke PRIVATE ole32 uuid)
  endif()
  if(CMAKE_SIZEOF_VOID_P EQUAL 8 AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64)$" AND NOT CMAKE_GENERATOR_PLATFORM MATCHES "^[Aa][Rr][Mm]64$")
    add_test(NAME BytepleatExplorerCommandsTests COMMAND BytepleatExplorerCommandsTests $<TARGET_FILE:BytepleatExplorerCommands>)
  endif()
endif()
